<script>
  // ── 要填的兩個常數（見上面的說明）──
  var APPSTORE_URL = "https://apps.apple.com/tw/app/yumi-badminton/id6799363645";
  var PLAY_URL     = "";   // Android 上架後填；空字串 = 顯示「即將上架」

  var id = new URLSearchParams(location.search).get("c") || "";
  // 球團 id 一定是 UUID —— 不是就當連結壞掉，也順便擋掉奇怪的注入
  var UUID_RE = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;
  if (!UUID_RE.test(id)) {
    document.getElementById("main").style.display = "none";
    document.getElementById("badlink").style.display = "block";
  }

  var deep = "yumi://club/" + id;
  var timer = null;

  function tryOpen(){
    document.getElementById("noapp").style.display = "none";
    location.href = deep;
    clearTimeout(timer);
    // 1.6 秒後這一頁還醒著 → 沒有 App 接手，亮出下載提示。
    // 有接手的話瀏覽器會被切到背景，timeout 到時 hidden 是 true。
    timer = setTimeout(function(){
      if (!document.hidden) document.getElementById("noapp").style.display = "block";
    }, 1600);
  }

  function goStore(){
    var isAndroid = /android/i.test(navigator.userAgent);
    if (isAndroid) {
      if (PLAY_URL) location.href = PLAY_URL;
      else alert("Android 版即將上架，敬請期待！\n目前可先用 iPhone 下載 Yumi。");
    } else {
      if (APPSTORE_URL) location.href = APPSTORE_URL;
      else alert("下載連結還沒設定好，請稍後再試。");
    }
  }

  document.getElementById("open").addEventListener("click", tryOpen);
  document.getElementById("retry").addEventListener("click", tryOpen);
  document.getElementById("store").addEventListener("click", goStore);
</script>
